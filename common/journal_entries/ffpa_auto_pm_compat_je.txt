	on_monthly_pulse = {
		effect = {
			debug_log = "FFPA_PM|ADAPTER_MONTHLY|Tech & Res adapter journal pulse"
			# Mirror Auto-Apply PMs' one/three-building budget for Tech & Res-only
			# building types, which have no upstream per-building scheduler.

				limit = {
					0 <= {
						if = {
							limit = { has_variable = zw_var_auto_pm_num_3 }

			else = { debug_log = "FFPA_PM|AUTOMATION_OIL|OFF" }

				limit = { has_variable = zw_var_auto_pm_transportation }

			else = { debug_log = "FFPA_PM|TRANSPORTATION|OFF" }

				limit = { has_variable = zw_var_auto_pm_slower_automation }
				debug_log = "FFPA_PM|AUTOMATION_FREQUENCY|TWICE_YEARLY"
			}
			else = { debug_log = "FFPA_PM|AUTOMATION_FREQUENCY|MONTHLY" }
			if = {
				limit = {
					any_scope_state = { zw_trigger_aaap_state_workforce_low = yes }
				}
				debug_log = "FFPA_PM|AUTOMATION_WORKFORCE_GATE|PASS"
			}
			else = { debug_log = "FFPA_PM|AUTOMATION_WORKFORCE_GATE|FAIL" }
			ffpa_update_auto_pm_trials = yes
			ffpa_process_new_techres_buildings = yes
			# The upstream automation journal only scans automation when labour is
			# scarce.  A bidirectional manager must also revisit states with ample
			# labour so an uneconomic automation method can be tested downward.
			if = {
				limit = {
					has_variable = zw_var_auto_pm_automation_basic
					OR = {
						NOT = { has_variable = zw_var_auto_pm_slower_automation }
						month = 0
						month = 6
					}
				}
				every_scope_state = { ffpa_process_state_techres_automation = yes }
			}
			if = {
				limit = {
					has_variable = zw_var_auto_pm_transportation
					OR = {
						NOT = { has_variable = zw_var_auto_pm_slower_automation }
						month = 0
						month = 6
					}
				}

			}
