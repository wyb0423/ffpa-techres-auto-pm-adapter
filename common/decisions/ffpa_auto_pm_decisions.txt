# Manual recovery path for an existing save. Normally the monthly country
# pulse adds the journal automatically; this decision makes recovery immediate.
ffpa_auto_pm_add_compat_journal = {
	is_shown = {
		is_player = yes
		NOT = { has_journal_entry = je_ffpa_auto_pm_compat }
	}

	when_taken = {
		add_journal_entry = {
			type = je_ffpa_auto_pm_compat
		}
	}
}
